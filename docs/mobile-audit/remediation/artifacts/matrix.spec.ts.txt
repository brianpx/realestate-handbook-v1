import {test, chromium, webkit, firefox, devices} from '@playwright/test';
import {signedInState, ADMIN_SESSION_ID} from './e2e/auth';
import {writeFileSync} from 'node:fs';
const out='/Users/mmbp/Projects/code/realestate-handbook-v1/docs/mobile-audit/remediation';
const widths=[320,360,375,390,402,430,480,768,820,1024,1280,1440];
const routes=['/','/preview','/pricing','/signin','/signup','/about','/support','/updates','/terms','/privacy','/refunds','/goodbye','/coming-soon','/auth/verify','/subscribe/confirm','/missing-audit-route','/handbook','/handbook/start','/handbook/plan','/handbook/review','/account',...['cover','pref',...Array.from({length:13},(_,i)=>`ch${i+1}`),...'abcdef'.split('').map(x=>`app-${x}`)].map(x=>'/handbook/'+x),'/admin','/admin/users','/admin/leads','/admin/audit'];
test('route and viewport evidence',async()=>{
 const records=[]; const availability=[];
 for(const [name,engine] of Object.entries({chromium,webkit,firefox})) {
 let browser;try{browser=await engine.launch();availability.push({name,status:'launched',version:browser.version()})}catch(e){availability.push({name,status:'unavailable',error:String(e)});continue}
 const context=await browser.newContext({...devices['iPhone 13'],storageState:signedInState([])});
 const page=await context.newPage();
 for(const route of routes){
 if(route.startsWith('/admin'))await context.addCookies(signedInState([],ADMIN_SESSION_ID).cookies);
 try{
 const response=await page.goto(route,{waitUntil:'domcontentloaded',timeout:60000});
 for(const width of widths){await page.setViewportSize({width,height:844});await page.waitForTimeout(30);
 const data=await page.evaluate(()=>({url:location.pathname,title:document.title,scrollWidth:document.documentElement.scrollWidth,innerWidth,bodyWidth:document.body.scrollWidth,offscreen:[...document.querySelectorAll('main *')].filter(e=>{const r=e.getBoundingClientRect();return r.width>0&&(r.right>innerWidth+1||r.left< -1)&&!e.closest('table,.table-wrap,[data-slot="table-container"]')}).slice(0,8).map(e=>({tag:e.tagName,cls:e.className,txt:e.textContent?.slice(0,65)})),smallInputs:[...document.querySelectorAll('input:not([type="hidden"]),textarea,select')].filter(e=>e.getBoundingClientRect().width>0&&parseFloat(getComputedStyle(e).fontSize)<16).map(e=>({id:e.id,size:getComputedStyle(e).fontSize})),h1:document.querySelectorAll('h1').length}));
 records.push({engine:name,route,width,status:response?.status(),...data});
 if([320,820,1440].includes(width)&&['/','/preview','/handbook/ch11','/handbook','/admin'].includes(route))await page.screenshot({path:`${out}/screenshots/${name}-${route.replaceAll('/','_')||'home'}-${width}.png`,fullPage:false});
 }
 }catch(e){records.push({engine:name,route,error:String(e)})}
 writeFileSync(out+'/artifacts/route-matrix.json',JSON.stringify({availability,records},null,2));
 }
 await browser.close();
 }
 writeFileSync(out+'/artifacts/route-matrix.json',JSON.stringify({availability,records},null,2));
});
