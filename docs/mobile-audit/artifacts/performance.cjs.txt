(async()=>{const browser=await chromium.launch();const rows=[];for(const route of ['/','/preview','/handbook/ch11']){const c=await browser.newContext({...devices['Pixel 7']});await c.addCookies([{name:'hb_session',value:'e2e-reader-session'.padEnd(43,'0'),domain:'localhost',path:'/'}]);const p=await c.newPage();const cdp=await c.newCDPSession(p);await cdp.send('Emulation.setCPUThrottlingRate',{rate:4});await cdp.send('Network.enable');await cdp.send('Network.emulateNetworkConditions',{offline:false,latency:150,downloadThroughput:200000,uploadThroughput:93750});await p.addInitScript(()=>{window.auditPerf={lcp:null,cls:0};new PerformanceObserver(l=>{for(const e of l.getEntries())window.auditPerf.lcp=e.startTime}).observe({type:'largest-contentful-paint',buffered:true});new PerformanceObserver(l=>{for(const e of l.getEntries())if(!e.hadRecentInput)window.auditPerf.cls+=e.value}).observe({type:'layout-shift',buffered:true})});await p.goto('http://localhost:3199'+route,{waitUntil:'networkidle',timeout:90000});await p.waitForTimeout(1500);rows.push({route,...await p.evaluate(()=>({...window.auditPerf,jsEncodedBytes:performance.getEntriesByType('resource').filter(e=>e.name.includes('.js')).reduce((n,e)=>n+e.encodedBodySize,0),jsDecodedBytes:performance.getEntriesByType('resource').filter(e=>e.name.includes('.js')).reduce((n,e)=>n+e.decodedBodySize,0),requests:performance.getEntriesByType('resource').length}))});await c.close()}fs.writeFileSync('/Users/mmbp/Projects/code/realestate-handbook-v1/docs/mobile-audit/artifacts/performance.json',JSON.stringify({method:'single cold browser-context navigation per route; production localhost; Chromium Pixel 7; CPU 4x; 150ms latency; 200000 bytes/s down; 93750 bytes/s up; LCP last candidate at capture; CLS raw sum excluding recent input (not session-window algorithm); no measured INP',rows},null,2));await browser.close()})();
